Topic Modeling Hotel Reviews with LDA

2025-04-08

Summary

This R Markdown document demonstrates the use of Latent Dirichlet Allocation (LDA) to extract underlying topics from a set of hotel reviews. The implementation involves loading and preprocessing text data, transforming it into a document-term matrix, and then fitting an LDA model using the topicmodels package. The results are visualized to explore common terms in each topic and investigate how topics vary across positive and negative review ratings.

# Load hotel reviews dataset
reviews <- read_csv("https://github.com/krisrs1128/stat436_f24/raw/refs/heads/main/data/reviews.csv")
## Rows: 5000 Columns: 3
## ── Column specification ────────────────────────────────────────────────────────
## Delimiter: ","
## chr (1): Review
## dbl (2): Rating, document
## 
## ℹ Use `spec()` to retrieve the full column specification for this data.
## ℹ Specify the column types or set `show_col_types = FALSE` to quiet this message.
# Tokenize text into words and remove stopwords
reviews_dtm <- reviews |>
  unnest_tokens(word, Review) |>                      # Converts reviews into individual lowercase words
  filter(!(word %in% stop_words$word)) |>             # Removes common stopwords
  count(document, word) |>                            # Counts word frequencies per review
  cast_dtm(document, word, n)                         # Converts to a DocumentTermMatrix for LDA
# Fit LDA model with 8 topics and a fixed seed for reproducibility
fit <- LDA(reviews_dtm, k = 8, control = list(seed = 123))
# Extract beta matrix (topic-term probabilities)
topics <- tidy(fit, matrix = "beta")
# Get top 10 terms per topic based on beta value
top_terms <- topics |>
  group_by(topic) |>
  slice_max(beta, n = 10) |>                          # Selects top 10 words per topic
  ungroup() |>
  arrange(topic)
# Plot the top terms per topic
top_terms |>
  mutate(term = reorder_within(term, beta, topic)) |>
  ggplot(aes(beta, term, fill = factor(topic))) +
  geom_col(show.legend = FALSE) +
  facet_wrap(~ topic, scales = "free") +              # One plot for each topic
  scale_y_reordered()

# Extract gamma matrix (document-topic probabilities)
reviews_topics <- tidy(fit, matrix = "gamma") |> 
  # Ensure the `document` column in `reviews` is a character to match the type in gamma matrix
  left_join(reviews |> mutate(document = as.character(document)), by = "document")
# Plot average topic probabilities by review rating
reviews_topics |>
  group_by(topic, Rating) |>
  summarize(avg_gamma = mean(gamma)) |>               # Computes average topic strength per rating
  ggplot(aes(Rating, avg_gamma, fill = factor(topic))) +
  geom_col(position = "dodge") +
  labs(y = "Avg. Topic Probability", x = "Rating")
